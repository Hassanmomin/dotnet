Practical-4

a. Design a Student Registration Form using the following ASP.NET controls:

i. Label and TextBox to accept Student Name

ii. RadioButton to select Gender (Male/Female)

iii. DropDownList to select Course (BSc IT, BSc CS, BCA)

iv. CheckBox for accepting Terms & Conditions

v. Calendar control to select Date

vi. Button control to submit the form

vii. Label control to display the output

StudentForm.aspx

StudentForm.aspx.cs
using System;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace StudentRegistration1
{
  public partial class StudentForm : System.Web.UI.Page
  {
    protected void Page_Load(object sender, EventArgs e)
    {

         }

         protected void btnSubmit_Click(object sender, EventArgs e)
         {
           string gender = "";

             if(rbMale.Checked)
             {
                gender = "Male";
             }
             else if(rbFemale.Checked)
             {
                gender = "Female";
             }
             if(chkTerms.Checked)
             {
                lblOutput.Text = "Student Name:" + txtName.Text +
                   "<br/>Gender:" + gender +
                   "<br/>Course:" + ddlCourse.SelectedValue +
                   "<br/>Date:" + calDate.SelectedDate.ToShortDateString();
             }
             else
             {
                lblOutput.Text = "Please accept Terms & Conditions";
             }

         }
     }
}
